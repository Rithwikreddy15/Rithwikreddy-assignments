vendor_product table :


create table Vendor_Product (
    VendorCode varchar(10),
    ProductCode varchar(10),
    primary key (VendorCode, ProductCode),
    foreign key (VendorCode) references Vendor(VendorCode),
    foreign key (ProductCode) references Product(ProductCode)
);

modifying the product table 

create table Product_3NF (
    ProductCode varchar(10) primary key,
    ProductDescription varchar(100),
    Price int,
    Grade char(1),
    foreign key (Grade) references Grade_Discount(Grade)
);
