#Activity 1

# creating database

create database ABC_MIS;
use ABC_MIS;

# creating employee table
create table employee (
    empid int primary key auto_increment,
    first_name varchar(100) not null,
    last_name varchar(100),
    contact varchar(10) not null unique,
    email varchar(50) not null unique,
    gender char(1),
    address varchar(100) not null,
    doj date not null,
    check (gender in ('M','F'))
);

# creating project table
create table project (
    prjid int primary key auto_increment,
    prj_name varchar(100) not null,
    start_date date not null,
    release_date date,
    status varchar(50),
    check (release_date > start_date),
    check (status in ('Released','In progress','Initiated','Not started'))
);

# creating emp_proj table

create table emp_proj (
    empid int,
    prjid int,
    primary key (empid, prjid),
    foreign key (empid) references employee(empid),
    foreign key (prjid) references project(prjid)
);

# creating salary table
create table salary (
    empid int,
    date date not null,
    amount float,
    remark varchar(100),
    primary key (empid, date),
    foreign key (empid) references employee(empid),
    check (amount >= 35000)
);

#activity 2

# tax calculation
select empid, amount, date, amount*0.105 as Tax
from salary;

# tax and net salary
select empid, amount, date,
amount*0.105 as Tax,
amount-(amount*0.105) as Net_Salary
from salary;

# net salary < 45000
select *, amount-(amount*0.105) as Net_Salary
from salary
having Net_Salary < 45000;

# doj between dates
select * from employee
where doj between '2013-04-01' and '2023-03-31';

# first name pattern
select * from employee
where first_name like 'P____';

# name condition
select * from employee
where first_name='Tom' or last_name='Altor';

# concat project
select concat(prjid,'_',prj_name) as prjid_prjname
from project;

# capital first letter
select concat(upper(left(prj_name,1)),lower(substring(prj_name,2))) 
from project;

# employee formatted name
select concat(empid,'_',upper(left(first_name,1)),
lower(substring(first_name,2,2)),'_',upper(left(last_name,1)))
from employee;

# date difference
select prjid, datediff(release_date,start_date) as days
from project;

# null handling
select empid, first_name, ifnull(last_name,'-') from employee;

# project full name
select prjid,
case
when prj_name='ACMS' then 'Account & Credit Management System'
when prj_name='OPS' then 'Order Processing System'
when prj_name='ERP' then 'Enterprise Resource Planning'
when prj_name='APMS' then 'Account & Payment Management System'
when prj_name='MIS' then 'Management Information System'
else 'Not Assigned'
end
from project;

#activity 4

# employees without project
select * from employee e
left join emp_proj ep on e.empid=ep.empid
where ep.prjid is null;

# projects without employees
select * from project p
left join emp_proj ep on p.prjid=ep.prjid
where ep.empid is null;

# employee project mapping
select * from employee e
join emp_proj ep on e.empid=ep.empid
join project p on ep.prjid=p.prjid;

# employees with at least one project

# salary less than average
select * from salary
